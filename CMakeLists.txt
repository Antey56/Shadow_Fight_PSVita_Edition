cmake_minimum_required(VERSION 3.16)
project(sf2_host_checks C)
set(CMAKE_C_STANDARD 99)
enable_testing()
add_compile_options(-Wall -Wextra -Werror)
include_directories(vita/include)
add_executable(elf_preflight tools/elf_preflight_main.c vita/src/elf_preflight.c)
add_executable(touch_test tests/touch_test.c vita/src/touch.c)
target_link_libraries(touch_test m)
add_executable(registry_test tests/registry_test.c vita/src/native_registry.c)
add_test(NAME touch COMMAND touch_test)
add_test(NAME native_registry COMMAND registry_test)
add_executable(reloc_test tests/reloc_test.c vita/src/arm_reloc.c)
add_test(NAME arm_relocations COMMAND reloc_test)
add_library(arm_dynamic_host SHARED vita/src/arm_dynamic.c)
set_target_properties(arm_dynamic_host PROPERTIES PREFIX "")
add_library(arm_link_host SHARED vita/src/arm_link.c vita/src/arm_dynamic.c vita/src/arm_reloc.c vita/src/arm_scope.c)
set_target_properties(arm_link_host PROPERTIES PREFIX "")
add_executable(scope_test tests/scope_test.c vita/src/arm_scope.c)
add_test(NAME symbol_scope COMMAND scope_test)
add_executable(android_log_test tests/android_log_test.c vita/src/android_log_adapter.c)
add_test(NAME android_logging COMMAND android_log_test)
add_library(android_asset_host SHARED vita/src/android_asset_adapter.c)
set_target_properties(android_asset_host PROPERTIES PREFIX "")
add_library(jni_objects_host SHARED vita/src/jni_objects.c vita/src/android_asset_adapter.c)
